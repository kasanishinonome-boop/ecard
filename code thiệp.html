🔒 Cách đặt mật khẩu, ảnh nền & khóa "Riêng tư":

1. Chèn ảnh nền Anime: Dán link ảnh từ Google (chuột phải vào ảnh → Sao chép địa chỉ hình ảnh) hoặc tên file cùng thư mục (vd: bg.jpg). Bạn cũng có thể chọn ảnh từ máy để xem thử ngay.

2. Gửi link online: Bấm Chép cấu hình code bên dưới và dán đè vào DEFAULT_CARD_DATA ở đầu file HTML rồi up lên Netlify / GitHub Pages.

3. Chuyển riêng tư sau khi họ đọc xong: Đổi password trong code thành mật khẩu mới (hoặc tắt trang trên host) là link cũ sẽ tự động khóa lại!

Hiệu ứng âm thanh dịu nhẹ bằng Tone.js (tổng hợp trực tiếp, không cần URL ngoài) async function playChimeSound(type = 'unlock') { try { if (typeof Tone === 'undefined') return; await Tone.start(); const synth = new Tone.PolySynth(Tone.Synth, { oscillator: { type: 'sine' }, envelope: { attack: 0.05, decay: 0.4, sustain: 0.2, release: 1.2 } }).toDestination(); synth.volume.value = -12; const now = Tone.now(); if (type === 'unlock') { // Hợp âm mở khóa trong trẻo synth.triggerAttackRelease('C5', '8n', now); synth.triggerAttackRelease('E5', '8n', now + 0.1); synth.triggerAttackRelease('G5', '4n', now + 0.2); synth.triggerAttackRelease('B5', '2n', now + 0.32); } else if (type === 'open') { // Giai điệu mở thiệp lãng mạn synth.triggerAttackRelease('F5', '8n', now); synth.triggerAttackRelease('A5', '8n', now + 0.12); synth.triggerAttackRelease('C6', '4n', now + 0.24); synth.triggerAttackRelease('E6', '2n', now + 0.38); } else if (type === 'error') { synth.triggerAttackRelease('E4', '16n', now); synth.triggerAttackRelease('Eb4', '8n', now + 0.1); } } catch (e) { // Bỏ qua nếu trình duyệt chặn audio tự động } } // 4. Hệ thống Canvas hiệu ứng cánh hoa pastel & đốm sáng lấp lánh const canvas = document.getElementById('petalCanvas'); const ctx = canvas.getContext('2d'); let petals = []; let animationFrameId = null; let gentleInterval = null; function resizeCanvas() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; } window.addEventListener('resize', resizeCanvas); resizeCanvas(); const PETAL_COLORS = [ 'rgba(206, 177, 235, 0.75)', // Tím oải hương 'rgba(51, 170, 238, 0.55)', // Xanh thiên thanh 'rgba(244, 194, 219, 0.75)', // Hồng phấn pastel 'rgba(255, 255, 255, 0.85)' // Ánh sáng trắng kem ]; function createPetal(fromTop = true) { return { x: Math.random() * canvas.width, y: fromTop ? -20 : Math.random() * canvas.height * 0.5, size: Math.random() * 7 + 5, speedY: Math.random() * 1.4 + 0.7, speedX: (Math.random() - 0.5) * 1.2, rotation: Math.random() * Math.PI * 2, rotationSpeed: (Math.random() - 0.5) * 0.04, swayOffset: Math.random() * Math.PI * 2, color: PETAL_COLORS[Math.floor(Math.random() * PETAL_COLORS.length)], isSparkle: Math.random() > 0.72 }; } function spawnPetalBurst(count = 25) { for (let i = 0; i < count; i++) { petals.push(createPetal(true)); } if (!animationFrameId) { animatePetals(); } } function startGentlePetals() { if (gentleInterval) clearInterval(gentleInterval); gentleInterval = setInterval(() => { if (petals.length < 40) { petals.push(createPetal(true)); if (!animationFrameId) animatePetals(); } }, 550); } function stopGentlePetals() { if (gentleInterval) clearInterval(gentleInterval); gentleInterval = null; } function animatePetals() { ctx.clearRect(0, 0, canvas.width, canvas.height); for (let i = petals.length - 1; i >= 0; i--) { const p = petals[i]; p.y += p.speedY; p.x += p.speedX + Math.sin(p.y * 0.02 + p.swayOffset) * 0.6; p.rotation += p.rotationSpeed; ctx.save(); ctx.translate(p.x, p.y); ctx.rotate(p.rotation); ctx.fillStyle = p.color; if (p.isSparkle) { // Vẽ đốm sáng ngôi sao 4 cánh ctx.beginPath(); const r = p.size * 0.6; ctx.moveTo(0, -r); ctx.quadraticCurveTo(0, 0, r, 0); ctx.quadraticCurveTo(0, 0, 0, r); ctx.quadraticCurveTo(0, 0, -r, 0); ctx.quadraticCurveTo(0, 0, 0, -r); ctx.fill(); } else { // Vẽ cánh hoa mềm mại ctx.beginPath(); ctx.ellipse(0, 0, p.size, p.size * 0.55, 0, 0, Math.PI * 2); ctx.fill(); } ctx.restore(); if (p.y > canvas.height + 30) { petals.splice(i, 1); } } if (petals.length > 0) { animationFrameId = requestAnimationFrame(animatePetals); } else { animationFrameId = null; } } // 7.
